cmake_minimum_required(VERSION 3.24)
project(oa_ui_frontend_state LANGUAGES CXX)
include(CTest)
add_library(oa-ui-frontend-state src/dispatcher.cpp src/app_modes.cpp src/main_menu.cpp src/game_entry.cpp src/initialization.cpp src/skirmish_ui.cpp src/map_selection.cpp src/oa_button_place.cpp)

target_include_directories(oa-ui-frontend-state PUBLIC include)
target_compile_features(oa-ui-frontend-state PUBLIC cxx_std_20)
if(TARGET oa-options)
  target_link_libraries(oa-ui-frontend-state PUBLIC oa-options)
endif()
target_link_libraries(oa-ui-frontend-state PUBLIC oa-core-types oa-data-match-rules PRIVATE oa-data-campaign oa-data-map-pack)
